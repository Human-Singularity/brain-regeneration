{{- /* Markdown images in news posts: ![alt](path "optional title"). Page-bundle images are resized to the column (max 1440px, webp), carry their dimensions so the page doesn't jump, and open full size in the lightbox. Remote images and formats Hugo can't resize (svg, gif) are passed through. */ -}} {{- $u := urls.Parse .Destination -}} {{- $res := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $res = or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- end -}} {{- $caption := .Title | default .Text -}} {{- if and $res (eq $res.ResourceType "image") (not (in (slice "svg" "gif") $res.MediaType.SubType)) -}} {{- $img := $res -}} {{- if gt $res.Width 1440 -}} {{- $img = $res.Resize "1440x webp" -}} {{- else -}} {{- $img = $res.Process "webp" -}} {{- end -}} {{- else -}} {{- end -}}
